--- [postgres@boardtrack_db] select pg_get_viewdef('v_lease_summary'::regclass) as view_definition
                                                    view_definition                                                    
-----------------------------------------------------------------------------------------------------------------------
  SELECT l.lease_id,                                                                                                  +
     l.tenant_id,                                                                                                     +
     t.full_name AS tenant,                                                                                           +
     p.property_name,                                                                                                 +
     r.room_number,                                                                                                   +
     l.start_date,                                                                                                    +
     l.end_date,                                                                                                      +
     l.monthly_rent,                                                                                                  +
     l.status,                                                                                                        +
     COALESCE(( SELECT sum(x.amount) AS sum                                                                           +
            FROM payment x                                                                                            +
           WHERE ((x.lease_id = l.lease_id) AND ((x.status)::text = 'COMPLETED'::text))), (0)::numeric) AS total_paid,+
     COALESCE(( SELECT sum(u.amount) AS sum                                                                           +
            FROM utility_bill u                                                                                       +
           WHERE (u.lease_id = l.lease_id)), (0)::numeric) AS utilities_billed                                        +
    FROM (((lease l                                                                                                   +
      JOIN tenant t ON ((t.tenant_id = l.tenant_id)))                                                                 +
      JOIN room r ON ((r.room_id = l.room_id)))                                                                       +
      JOIN property p ON ((p.property_id = r.property_id)));
